bash <<'BASH'
set -euo pipefail
cd /home/jovyan/winstonxcai/flash-optimizations
git pull --ff-only origin remnant

git lfs install --local

# Select the newest launch for each Packed mode by its timestamped name.
# Include all completed capture windows (c24 and c28 when present).
traces=()
for mode in triton kernel; do
  profile_dir=$(find logs -maxdepth 1 -type d -name "nsys-${mode}-*" -print |
    LC_ALL=C sort | tail -n 1)
  if [[ -z "$profile_dir" ]]; then
    echo "No Nsight launch directory found for $mode" >&2
    exit 1
  fi
  mapfile -d '' mode_traces < <(
    find "$profile_dir" -maxdepth 1 -type f -name 'server.*.nsys-rep' -print0
  )
  if ((${#mode_traces[@]} < 2)); then
    echo "Expected at least two completed reports in $profile_dir" >&2
    exit 1
  fi
  traces+=("${mode_traces[@]}")
done

# The repository already tracks *.nsys-rep through LFS.
for trace in "${traces[@]}"; do
  if [[ $(git check-attr filter -- "$trace") != *': filter: lfs' ]]; then
    echo "Missing LFS rule for $trace; refusing to stage a large Git blob" >&2
    exit 1
  fi
done

du -h -- "${traces[@]}"
git add -f -- "${traces[@]}"
git lfs status
git diff --cached --stat -- "${traces[@]}"
if ! git diff --cached --quiet -- "${traces[@]}"; then
  git commit --only -m "Store Packed Triton and Kernel Nsight traces in LFS" \
    -- "${traces[@]}"
fi
git push origin remnant
BASH
